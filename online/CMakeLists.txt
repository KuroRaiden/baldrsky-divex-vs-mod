cmake_minimum_required(VERSION 3.20)
project(BaldrOnline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_compile_options(/utf-8)
set(NO_ASIO_BUILD ON CACHE BOOL "Use the AutoPunch-compatible Winsock adapter" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${GEKKO_PATH}" gekko)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${ONLINE_OUT}")
set(DETOURS "${AUTOPUNCH_PATH}/autopunch-inject/detours/src")
add_library(BaldrAutoPunch MODULE "${ONLINE_OUT}/autopunch_start.c"
 "${DETOURS}/detours.cpp" "${DETOURS}/modules.cpp" "${DETOURS}/disasm.cpp" "${DETOURS}/image.cpp" "${DETOURS}/creatwth.cpp"
 "${DETOURS}/disolx86.cpp" "${DETOURS}/disolx64.cpp" "${DETOURS}/disolia64.cpp" "${DETOURS}/disolarm.cpp" "${DETOURS}/disolarm64.cpp")
target_include_directories(BaldrAutoPunch PRIVATE "${DETOURS}")
target_compile_definitions(BaldrAutoPunch PRIVATE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS AUTOPUNCH_VERSION=0.0.0.42)
target_link_libraries(BaldrAutoPunch PRIVATE shlwapi ws2_32 shell32)
target_link_options(BaldrAutoPunch PRIVATE /EXPORT:StartAutoPunch=_StartAutoPunch@0)
set_target_properties(BaldrAutoPunch PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${ONLINE_OUT}")
add_executable(test_online "${ROOT}/src/test_online.cpp")
target_compile_options(test_online PRIVATE /UNDEBUG)
target_link_libraries(test_online PRIVATE GekkoNet ws2_32 winhttp)
add_executable(test_online_chat "${ROOT}/src/test_online_chat.cpp")
target_compile_options(test_online_chat PRIVATE /UNDEBUG)
add_executable(BaldrVSLauncherOnline WIN32 "${ROOT}/src/vs_launcher_menu.cpp" "${ROOT}/src/launcher.rc"
 "${IMGUI_PATH}/imgui.cpp" "${IMGUI_PATH}/imgui_draw.cpp" "${IMGUI_PATH}/imgui_tables.cpp" "${IMGUI_PATH}/imgui_widgets.cpp"
 "${IMGUI_PATH}/backends/imgui_impl_win32.cpp" "${IMGUI_PATH}/backends/imgui_impl_dx11.cpp")
target_include_directories(BaldrVSLauncherOnline PRIVATE "${IMGUI_PATH}" "${IMGUI_PATH}/backends" "${ROOT}/assets")
target_compile_definitions(BaldrVSLauncherOnline PRIVATE VS_NATIVE_START VS_ONLINE WIN32_LEAN_AND_MEAN IMGUI_IMPL_WIN32_DISABLE_GAMEPAD)
target_link_libraries(BaldrVSLauncherOnline PRIVATE GekkoNet d3d11 d3dcompiler user32 gdi32 dwmapi advapi32 ws2_32 winhttp)
# Built explicitly for local engine research; never copied into a launcher build.
add_library(NativeReplayAudit MODULE EXCLUDE_FROM_ALL "${ROOT}/src/vs_inject.cpp" "${ROOT}/src/vs_rounds_v11c.c")
set_source_files_properties("${ROOT}/src/vs_rounds_v11c.c" PROPERTIES LANGUAGE CXX)
target_compile_options(NativeReplayAudit PRIVATE /TP)
target_compile_definitions(NativeReplayAudit PRIVATE VS_NATIVE_START VS_SPLIT_SCREEN ROUND_NATIVE_PLUGIN ONLINE_NATIVE_REPLAY_AUDIT)
target_link_libraries(NativeReplayAudit PRIVATE kernel32 user32)
set_target_properties(NativeReplayAudit PROPERTIES LIBRARY_OUTPUT_DIRECTORY_RELEASE "${ONLINE_OUT}")
target_link_options(NativeReplayAudit PRIVATE /EXPORT:InitializeVS=_InitializeVS@4 /EXPORT:StartVSPlugin=_StartVSPlugin@0)
